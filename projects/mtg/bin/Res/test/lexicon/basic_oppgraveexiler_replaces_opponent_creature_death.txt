FIRSTMAIN

inplay:Lexicon Opp Grave Exiler Witness

manapool:{2}{B}{R}

goto end
[ASSERT]
END
[PLAYER1]
inplay:Lexicon Opp Grave Exiler Witness

manapool:{0}
[PLAYER2]
exile:Hill Giant
[END]
