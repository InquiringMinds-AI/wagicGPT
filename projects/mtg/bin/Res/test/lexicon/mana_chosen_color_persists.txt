# {chosencolor}: Coldsteel Heart's chosen color persists across turns and
# the granted {T}: add {chosencolor} produces it. Cast protocol: choice 0
# twice (accept as-enters choice, then color pick; 0 = green).
# CS-020 regression: the later activation is a DIRECT click - no menu. If a
# stale "Choose a color" chooser ever persists again it would open a menu
# here, eat the following commands, and fail this test.
[INIT]
FIRSTMAIN
[PLAYER1]
hand:coldsteel heart
manapool:{2}
[PLAYER2]
[DO]
coldsteel heart
choice 0
choice 0
goto end
goto firstmain
goto end
goto firstmain
coldsteel heart
[ASSERT]
FIRSTMAIN
[PLAYER1]
inplay:coldsteel heart
manapool:{G}
[PLAYER2]
[END]
