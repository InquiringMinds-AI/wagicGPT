# removes it at the next upkeep, then casts the spell without its mana cost.

FIRSTMAIN
[PLAYER1]

manapool:{R}
[PLAYER2]

[DO]

goto end
goto upkeep
goto end
goto upkeep
next
p2
[ASSERT]
UPKEEP
[PLAYER1]
graveyard:rift bolt
manapool:{0}
[PLAYER2]
life:17
[END]
