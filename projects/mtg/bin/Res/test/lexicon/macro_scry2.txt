FIRSTMAIN
[PLAYER1]
inplay:Lexicon Scry 2 Macro Witness
library:Forest,Island,Mountain
life:20
[PLAYER2]
[DO]
Lexicon Scry 2 Macro Witness
[ASSERT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Scry 2 Macro Witness
library:*,*,*
life:21
[PLAYER2]
[END]
