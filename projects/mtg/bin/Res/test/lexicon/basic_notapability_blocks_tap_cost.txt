# notapability suppresses an activated ability whose cost contains {T}: the
# witness's {T}:damage ability cannot be used, so it stays untapped and deals
# no damage. (Script does goto end, so assert the END phase it lands on.)
[INIT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon NoTapAbility Witness
[PLAYER2]
life:20
[DO]
Lexicon NoTapAbility Witness
goto end
[ASSERT]
END
[PLAYER1]
inplay:Lexicon NoTapAbility Witness
tappedinplay:0
[PLAYER2]
life:20
[END]
