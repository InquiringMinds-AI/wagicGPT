# The active player casts the bolt at Phyrexian Unlife's controller.

FIRSTMAIN
[PLAYER1]

[PLAYER2]
inplay:Phyrexian Unlife

[DO]
Lightning Bolt
p2
goto firstmain
[ASSERT]
gameover:none
FIRSTMAIN
[PLAYER1]
graveyard:Lightning Bolt
manapool:{0}
[PLAYER2]
inplay:Phyrexian Unlife
life:-2
[END]
