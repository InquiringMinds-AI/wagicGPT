# canblocktapped allows a tapped creature to be selected as a blocker.
[INIT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Can Block Tapped Witness
[PLAYER2]
inplay:Raging Goblin
[DO]
Lexicon Can Block Tapped Witness
goto end
goto firstmain
goto attackers
Raging Goblin
goto blockers

goto combatdamage
next
[ASSERT]

[PLAYER1]
inplay:Lexicon Can Block Tapped Witness

[PLAYER2]

[END]
