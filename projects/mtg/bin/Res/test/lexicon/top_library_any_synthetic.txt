# canplayfromlibrarytop lets any card type be cast from the top of library.
[INIT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Synthetic Top Any Permission
library:Grizzly Bears
manapool:{1}{G}
[PLAYER2]
[DO]
Grizzly Bears
[ASSERT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Synthetic Top Any Permission,Grizzly Bears
manapool:{0}
[PLAYER2]
life:20
[END]
