[INIT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Synthetic Top Aura Equipment Permission

manapool:{1}
[PLAYER2]
[DO]
Bonesplitter
[ASSERT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Synthetic Top Aura Equipment Permission,Bonesplitter
manapool:{0}
[PLAYER2]
life:20
[END]
