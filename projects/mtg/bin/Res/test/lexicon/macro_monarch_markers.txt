# Monarch marker mechanic, observed DIRECTLY (no `&& life:1` rider -- the parser
# would fold it into the else branch).
#
# The Monarch is a marker token carrying `becomesmonarch controller` and
# `@each my end restriction{pmonarch>0}: draw:1 controller`: its controller
# becomes the monarch and draws a card at the beginning of their end step.
#
# _MONARCH_CONTROLLER_ creates The Monarch on player 1's battlefield and player 1
# becomes the monarch. Advancing to player 1's end step fires the monarch draw --
# the top library card (Mountain) is drawn (library 3 -> 2, +1 hand). The marker
# is asserted by NAME (marker tokens name-match the zone assert; full P/T creature
# tokens are id-matched and need a `*` wildcard).
#
# The opponent variant (_MONARCH_OPPONENT_, verified separately during authoring)
# creates The Monarch on player 2's battlefield instead; it is not combined here
# because both macros gate on the single shared `type(The Monarch|battlefield)`
# check and would contend for the same marker.
[INIT]
FIRSTMAIN
[PLAYER1]
inplay:Lexicon Monarch Controller Macro Witness
library:Forest,Island,Mountain
[PLAYER2]
[DO]
Lexicon Monarch Controller Macro Witness
goto end
[ASSERT]
END
[PLAYER1]
inplay:Lexicon Monarch Controller Macro Witness,The Monarch
library:Forest,Island
hand:Mountain
[PLAYER2]
[END]
