# --- 613 sharp discriminator (ability-loss vs a creature-based lord).
# Var Humbling Field is Humility-class: every creature becomes base 1/1 AND
# loses all abilities (layers 6 + 7b). Var Anthem Beast is a CREATURE whose
# printed ability is a lord giving every creature +1/+1 (layer 7c, but it is an
# ABILITY of a creature). Under Humility the Beast loses that ability, so no
# +1/+1 exists to apply. CR result: all creatures are 1/1. The audit flags
# ALoseAbilities as a "hack" that exempts lord-granted abilities; if so the
# +1/+1 survives and creatures are 2/2 (CR-wrong).
[card]
id=2100000720
name=Var Humbling Field
mana={2}{W}{W}
type=Enchantment
auto=lord(creature) becomes(Creature,1/1)
auto=lord(creature) loseabilities
text=All creatures have base power and toughness 1/1 and lose all abilities.
[/card]
[card]
id=2100000721
name=Var Anthem Beast
mana={2}{G}
type=Creature
subtype=Beast
power=3
toughness=3
auto=lord(creature) 1/1
text=Creatures you control get +1/+1 (a creature-based lord).
[/card]

# --- 613 dependency discriminator (layer 4 type-change feeding a layer-7c
# type-conditional pump). Var Graft Field makes every creature also a Zombie
# (a type change). Var Zombie Muster gives +2/+2 to Zombies. The pump DEPENDS
# on the graft's output. CR (613.7 dependency): the type change always applies
# first, so Var Measure Golem is a Zombie and gets +2/+2 = 6/6 (deals 6),
# regardless of which effect began first. If the engine evaluates the pump
# before the type-change in a frame, the golem is not yet a Zombie and misses
# the +2/+2 (stays 4/4, deals 4) -> order-dependent = 613 dependency PROVEN.
[card]
id=2100000730
name=Var Graft Field
mana={2}{U}
type=Enchantment
auto=lord(creature) transforms((Zombie))
text=Every creature is also a Zombie.
[/card]
[card]
id=2100000731
name=Var Zombie Muster
mana={2}{B}
type=Enchantment
auto=lord(creature[zombie]) 2/2
text=Zombies get +2/+2.
[/card]
